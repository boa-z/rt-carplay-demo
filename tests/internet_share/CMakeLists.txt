cmake_minimum_required(VERSION 3.16)
project(carplay_internet_share C)
get_filename_component(BUNDLED_SDK "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(SDK_ROOT "${BUNDLED_SDK}" CACHE PATH "Luban-Lite integration checkout")
set(LWIP "${SDK_ROOT}/kernel/rt-thread/components/net/lwip/lwip-2.1.2")
file(GLOB CORE "${LWIP}/src/core/*.c" "${LWIP}/src/core/ipv4/*.c")
add_executable(test_internet_route route.c ${CORE})
target_include_directories(test_internet_route PRIVATE support .)
target_compile_options(test_internet_route PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter -g -idirafter "${LWIP}/src/include")
target_link_options(test_internet_route PRIVATE -Wl,--wrap=mem_calloc)
option(SANITIZE "Use ASan/UBSan" ON)
if(SANITIZE)
    target_compile_options(test_internet_route PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
    target_link_options(test_internet_route PRIVATE -fsanitize=address,undefined -no-pie)
endif()
enable_testing()
add_test(NAME actual_lwip_ethernet_ap_route COMMAND test_internet_route)
add_executable(test_internet_supervisor supervisor.c ../../app/network_manager.c ${CORE})
target_include_directories(test_internet_supervisor PRIVATE support . ../../app)
target_compile_options(test_internet_supervisor PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter -g -idirafter "${LWIP}/src/include")
if(SANITIZE)
    target_compile_options(test_internet_supervisor PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
    target_link_options(test_internet_supervisor PRIVATE -fsanitize=address,undefined -no-pie)
endif()
add_test(NAME actual_share_supervisor COMMAND test_internet_supervisor)
