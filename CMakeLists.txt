cmake_minimum_required(VERSION 3.16)
project(rt_carplay_demo C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(RTCP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../rt-carplay" CACHE PATH "Receiver source")
set(LVGL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../lvgl" CACHE PATH "LVGL source")
set(RTCP_BUILD_TESTS OFF CACHE BOOL "Build Receiver tests")
set(RTCP_BUILD_SIMULATOR OFF CACHE BOOL "Build Receiver simulator")
option(CARPLAY_AIC_MPP "Use the ArtInChip framebuffer and H.264 decoder" OFF)
option(CARPLAY_ALSA "Build the ALSA backend" OFF)
option(CARPLAY_USB "Build the Linux USB host transport" OFF)
option(CARPLAY_LVGL "Build the LVGL launcher" OFF)
option(CARPLAY_BLUEZ "Build the Classic Bluetooth iAP2 transport" OFF)

find_package(Threads REQUIRED)
add_subdirectory("${RTCP_ROOT}" rt-carplay)
add_executable(rt-carplay-demo platform/linux/main.c platform/linux/persistence.c
    platform/linux/mfi.c platform/linux/gnss.c platform/linux/boot_trace.c app/receiver_session.c app/connection_retry.c
    app/phone_track.c)
target_include_directories(rt-carplay-demo PRIVATE app platform/linux)
target_compile_options(rt-carplay-demo PRIVATE -Wall -Wextra -Werror)
target_link_libraries(rt-carplay-demo PRIVATE rtcp_core rtcp_port_linux Threads::Threads)

if(CARPLAY_ALSA OR CARPLAY_USB OR CARPLAY_BLUEZ)
    find_package(PkgConfig REQUIRED)
endif()
if(CARPLAY_BLUEZ)
    pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
    target_sources(rt-carplay-demo PRIVATE platform/linux/wireless.c)
    target_compile_definitions(rt-carplay-demo PRIVATE CARPLAY_BLUEZ)
    target_link_libraries(rt-carplay-demo PRIVATE PkgConfig::GIO)
endif()
if(CARPLAY_ALSA)
    pkg_check_modules(ALSA REQUIRED IMPORTED_TARGET alsa)
    target_sources(rt-carplay-demo PRIVATE platform/linux/audio.c)
    target_compile_definitions(rt-carplay-demo PRIVATE CARPLAY_ALSA)
    target_link_libraries(rt-carplay-demo PRIVATE PkgConfig::ALSA)
endif()
if(CARPLAY_USB)
    pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
    target_sources(rt-carplay-demo PRIVATE platform/linux/wired.c app/usb_pair_cache.c)
    target_compile_definitions(rt-carplay-demo PRIVATE CARPLAY_USB)
    target_link_libraries(rt-carplay-demo PRIVATE PkgConfig::LIBUSB)
endif()
if(CARPLAY_AIC_MPP)
    find_path(AIC_MPP_INCLUDE_DIR mpp_decoder.h PATH_SUFFIXES local/include REQUIRED)
    find_library(AIC_MPP_LIBRARY NAMES mpp_decoder PATH_SUFFIXES local/lib REQUIRED)
    target_sources(rt-carplay-demo PRIVATE platform/linux/video.c app/fusion_display.c)
    target_include_directories(rt-carplay-demo PRIVATE "${AIC_MPP_INCLUDE_DIR}")
    target_compile_definitions(rt-carplay-demo PRIVATE CARPLAY_AIC_MPP RTCARPLAY_FUSION_POSIX)
    target_link_libraries(rt-carplay-demo PRIVATE "${AIC_MPP_LIBRARY}")
    set_target_properties(rt-carplay-demo PROPERTIES INSTALL_RPATH "\$ORIGIN/../local/lib")
    target_link_options(rt-carplay-demo PRIVATE LINKER:--disable-new-dtags)
endif()
if(CARPLAY_LVGL)
    set(LV_CONF_PATH "")
    set(BUILD_SHARED_LIBS OFF)
    set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
    set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "Skip LVGL examples")
    set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "Skip LVGL demos")
    set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "Skip internal ThorVG")
    add_subdirectory("${LVGL_ROOT}" lvgl EXCLUDE_FROM_ALL)
    target_include_directories(lvgl PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/platform/linux")
    target_sources(rt-carplay-demo PRIVATE platform/linux/launcher.c
        ui/carplay_ui_theme.c ui/launcher_layout.c ui/settings_navigation.c ui/connectivity_page.c
        ui/system_shell.c ui/ui_language.c ui/icons/carplay_icons.c ui/fonts/carplay_ui_17.c
        ui/vehicle_instruments.c ui/fonts/carplay_ui_21.c
        ui/fonts/carplay_display_58.c ui/fonts/carplay_display_220.c
        ui/fonts/carplay_ui_14.c ui/fonts/carplay_ui_15.c ui/fonts/carplay_ui_18.c ui/fonts/carplay_ui_34.c
        ui/fonts/carplay_cjk_16.c ui/fonts/carplay_cjk_18.c
        ui/fonts/carplay_cjk_24.c ui/fonts/carplay_cjk_30.c ui/fonts/carplay_cjk_full_18.c)
    target_compile_definitions(rt-carplay-demo PRIVATE CARPLAY_LVGL)
    target_include_directories(rt-carplay-demo PRIVATE ui)
    target_link_libraries(rt-carplay-demo PRIVATE lvgl m)
    if(CARPLAY_AIC_MPP)
        target_sources(rt-carplay-demo PRIVATE ui/fusion_dashboard.c ui/fonts/carplay_display_115.c)
    endif()
endif()

install(TARGETS rt-carplay-demo RUNTIME DESTINATION bin)
